📝 Getting Access to the Application:

{{- if eq .Values.frontend.service.type "NodePort" }}

1. Frontend (NodePort):
   export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "todo-app.fullname" . }}-frontend)
   export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")

   Access via: http://$NODE_IP:$NODE_PORT

   Or using Minikube:
   minikube service {{ include "todo-app.fullname" . }}-frontend --url

{{- else }}

1. Frontend:
   kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "todo-app.fullname" . }}-frontend {{ .Values.frontend.service.port }}:{{ .Values.frontend.service.port }}

   Access via: http://localhost:{{ .Values.frontend.service.port }}

{{- end }}

2. Backend API:
   kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "todo-app.fullname" . }}-backend {{ .Values.backend.service.port }}:{{ .Values.backend.service.port }}

   Access via: http://localhost:{{ .Values.backend.service.port }}
   API Docs: http://localhost:{{ .Values.backend.service.port }}/docs
